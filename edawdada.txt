-- PROPER FIX: Scale ONLY the ZyronX UI, not the game
-- ============================================================================
task.spawn(function()
    task.wait(0.5) -- Wait for UI to initialize
    
    local playerGui = Players.LocalPlayer:WaitForChild("PlayerGui")
    
    -- Find the ZyronX ScreenGui specifically
    local zyronxGui = nil
    for _, gui in pairs(playerGui:GetChildren()) do
        if gui:IsA("ScreenGui") then
            -- Look for the main UI frame
            local hasLargeFrame = false
            for _, child in pairs(gui:GetDescendants()) do
                if child:IsA("Frame") and child.AbsoluteSize.X > 800 then
                    hasLargeFrame = true
                    break
                end
            end
            
            if hasLargeFrame then
                zyronxGui = gui
                print("[SCALE] Found ZyronX UI:", gui.Name)
                break
            end
        end
    end
    
    if zyronxGui then
        -- Remove any existing UIScale to start fresh
        for _, child in pairs(zyronxGui:GetChildren()) do
            if child:IsA("UIScale") then
                child:Destroy()
            end
        end
        
        -- Add a NEW UIScale that only affects this ScreenGui
        local uiScale = Instance.new("UIScale")
        uiScale.Scale = 0.5 -- 50% size
        uiScale.Parent = zyronxGui
        
        print("[SCALE] Applied 0.5 scale to ZyronX UI only")
        print("[SCALE] Game UI should remain unaffected")
        
        -- Verify game UI is not scaled
        task.wait(0.5)
        for _, gui in pairs(playerGui:GetChildren()) do
            if gui ~= zyronxGui and gui:IsA("ScreenGui") then
                -- Make sure other GUIs don't have unwanted scaling
                local otherScale = gui:FindFirstChildOfClass("UIScale")
                if otherScale and otherScale.Scale ~= 1 then
                    warn("[WARNING] Found scaled UI that isn't ZyronX:", gui.Name, "Scale:", otherScale.Scale)
                end
            end
        end
    else
        warn("[SCALE] Could not find ZyronX UI to scale")
    end
end)
