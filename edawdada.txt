-- Test 8: Try to add ESPPreview in a separate thread
print("[TEST 8] Attempting to add ESPPreview in async mode...")
task.spawn(function()
    print("[TEST 8] [ASYNC] Starting ESPPreview creation...")
    
    local espSuccess, espResult = pcall(function()
        print("[TEST 8] [ASYNC] About to call Window:ESPPreview...")
        
        local preview = Window:ESPPreview({
            Name = "Character Preview"
        })
        
        print("[TEST 8] [ASYNC] Window:ESPPreview returned!")
        print("[TEST 8] [ASYNC] Preview type:", type(preview))
        print("[TEST 8] [ASYNC] Preview value:", tostring(preview))
        
        if not preview then
            warn("[TEST 8] [ASYNC] Preview is nil!")
            return nil
        end
        
        -- Wait a moment for initialization
        print("[TEST 8] [ASYNC] Waiting for preview to initialize...")
        task.wait(1)
        
        local Players = game:GetService("Players")
        local LocalPlayer = Players.LocalPlayer
        
        print("[TEST 8] [ASYNC] LocalPlayer:", tostring(LocalPlayer))
        
        if preview and type(preview) == "table" then
            print("[TEST 8] [ASYNC] Listing preview methods:")
            local methodCount = 0
            for k, v in pairs(preview) do
                if type(v) == "function" then
                    print("[TEST 8] [ASYNC]   - " .. tostring(k))
                    methodCount = methodCount + 1
                end
            end
            print("[TEST 8] [ASYNC] Total methods found:", methodCount)
        end
        
        if LocalPlayer then
            print("[TEST 8] [ASYNC] Waiting for character...")
            local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
            print("[TEST 8] [ASYNC] Character found:", character.Name)
            
            if preview and type(preview) == "table" then
                -- Try all possible methods
                local methods = {
                    "SetCharacter", "UpdateCharacter", "SetModel", "Update",
                    "SetPreview", "LoadCharacter", "ShowCharacter", "Refresh",
                    "Set", "Load", "UpdatePreview"
                }
                
                print("[TEST 8] [ASYNC] Trying to set character...")
                local success = false
                for _, methodName in ipairs(methods) do
                    if preview[methodName] then
                        print("[TEST 8] [ASYNC] Trying method:", methodName)
                        local ok, err = pcall(function()
                            preview[methodName](preview, character)
                        end)
                        if ok then
                            print("[TEST 8] [ASYNC] ✓ Character set via", methodName)
                            success = true
                            break
                        else
                            print("[TEST 8] [ASYNC] ✗", methodName, "failed:", tostring(err))
                        end
                    end
                end
                
                if not success then
                    print("[TEST 8] [ASYNC] ⚠ No working method found")
                end
            end
        else
            print("[TEST 8] [ASYNC] ⚠ LocalPlayer not found")
        end
        
        return preview
    end)
    
    print("[TEST 8] [ASYNC] pcall completed. Success:", espSuccess)
    
    if not espSuccess then
        warn("[ERROR] ESPPreview error:", tostring(espResult))
        print("[TEST 8] ✗ ESPPreview failed")
    else
        print("[TEST 8] ✓ ESPPreview process completed")
    end
end)
print("[TEST 8] ✓ ESPPreview started in background thread")
task.wait(0.1) -- Brief pause to let it initialize
